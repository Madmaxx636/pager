cmake_minimum_required(VERSION 3.21)
project(pager-native VERSION 0.1.4 LANGUAGES CXX)
add_compile_definitions(PAGER_NATIVE_VERSION="${PROJECT_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network)
qt_standard_project_setup(REQUIRES 6.4)

# The Rust engine (built by build-core.sh) sits next to the program.
set(PAGER_CORE_DIR "${CMAKE_SOURCE_DIR}/build-core" CACHE PATH "Folder with libpager_core.so")
add_library(pager_core SHARED IMPORTED)
if(WIN32)
  set_target_properties(pager_core PROPERTIES IMPORTED_LOCATION "${PAGER_CORE_DIR}/pager_core.dll" IMPORTED_IMPLIB "${PAGER_CORE_DIR}/libpager_core.dll.a")
else()
  set_target_properties(pager_core PROPERTIES IMPORTED_LOCATION "${PAGER_CORE_DIR}/libpager_core.so")
endif()

qt_add_executable(pager-native src/main.cpp src/engine.cpp src/store.cpp src/models.cpp src/appearance.cpp src/appsettings.cpp src/emoji.cpp src/fileinfo.cpp src/selfupdate.cpp src/rich.cpp src/names.cpp src/notifypolicy.cpp)
target_include_directories(pager-native PRIVATE include src)
target_link_libraries(pager-native PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 pager_core)
# The program finds its engine next to itself. In the build folder a copy is put there after each build, so no build path is baked into the program.
set_target_properties(pager-native PROPERTIES INSTALL_RPATH "$ORIGIN:$ORIGIN/../lib" BUILD_WITH_INSTALL_RPATH ON)
if(NOT WIN32)
  add_custom_command(TARGET pager-native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PAGER_CORE_DIR}/libpager_core.so" "$<TARGET_FILE_DIR:pager-native>/libpager_core.so")
endif()

qt_add_resources(pager-native icon PREFIX / FILES pager.svg emoji-names.tsv emoji-groups.tsv palettes.tsv)
qt_add_qml_module(pager-native URI Pager VERSION 1.0 RESOURCE_PREFIX / QML_FILES
  qml/Main.qml qml/Login.qml qml/RoomList.qml qml/Chat.qml qml/Avatar.qml qml/Bubble.qml qml/Recovery.qml qml/EmojiPicker.qml qml/PollCard.qml qml/PollDialog.qml qml/Settings.qml qml/Accounts.qml qml/QrView.qml qml/Networks.js qml/Admin.qml qml/SmoothWheel.qml qml/AudioBubble.qml qml/ImageViewer.qml qml/ForwardDialog.qml qml/InfoPanel.qml qml/IconButton.qml qml/PSwitch.qml qml/NewChat.qml qml/SearchDialog.qml qml/WhenDialog.qml qml/SGroup.qml qml/SToggle.qml qml/SChoice.qml qml/SSlider.qml qml/SList.qml qml/SAction.qml qml/SText.qml qml/Shortcuts.js)
if(WIN32)
  set_target_properties(pager-native PROPERTIES WIN32_EXECUTABLE ON)
endif()
install(TARGETS pager-native RUNTIME DESTINATION bin)
